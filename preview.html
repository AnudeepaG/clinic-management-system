👤usersCore Auth
PK id BIGINT AI
username VARCHAR(100) UNIQUE
password VARCHAR(255)
email VARCHAR(150) UNIQUE
role ENUM(ADMIN,DOCTOR,PATIENT)
enabled BOOLEAN DEFAULT TRUE
created_at DATETIME
👨‍⚕️doctorsDoctor Profile
PK id BIGINT AI
FK user_id → users UNIQUE
full_name VARCHAR(150)
specialty VARCHAR(100)
license_number VARCHAR(50)
available_slots TEXT (CSV)
slots_per_day INT DEFAULT 20
🧑patientsPatient Profile
PK id BIGINT AI
FK user_id → users UNIQUE
full_name VARCHAR(150)
date_of_birth DATE
gender VARCHAR(10)
phone VARCHAR(20)
blood_group VARCHAR(5)
emergency_contact VARCHAR(20)
📅appointmentsBooking
PK id BIGINT AI
FK doctor_id → doctors
FK patient_id → patients
appointment_date DATE
time_slot TIME
status ENUM(SCHEDULED…)
🔑 UNIQUE(doctor_id, date, slot)
💊prescriptionsMedical Record
PK id BIGINT AI
FK patient_id → patients
FK doctor_id → doctors
FK appointment_id → appointments
disease VARCHAR(200)
medications TEXT
dosage_instructions TEXT
follow_up_date DATETIME
timestamp DATETIME NOT NULL
FK editable_by_doctor_id → doctors
